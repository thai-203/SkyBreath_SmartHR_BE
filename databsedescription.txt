CREATE TABLE `employees` (
  `id` int, -- Khóa chính, ID nhân viên
  `employee_code` varchar, -- Mã số nhân viên (VD: NV001)
  `user_id` int, -- FK -> users.id (Tài khoản hệ thống)
  `full_name` varchar, -- Họ tên đầy đủ nhân viên (dùng để tìm kiếm)
  `date_of_birth` date, -- Ngày sinh nhân viên
  `gender` varchar, -- Giới tính (MALE/FEMALE/OTHER)
  `marital_status` varchar, -- Tình trạng hôn nhân (SINGLE/MARRIED...)
  `national_id` varchar, -- Số CCCD/CMND
  `tax_code` varchar, -- Mã số thuế cá nhân
  `personal_email` varchar, -- Email cá nhân
  `company_email` varchar, -- Email công ty
  `phone_number` varchar, -- Số điện thoại
  `department_id` int, -- FK -> departments.id (Mã phòng ban hiện tại)
  `position_id` int, -- FK -> positions.id (Mã chức vụ/chức danh)
  `job_grade_id` int, -- FK -> job_grades.id (Mã bậc lương)
  `direct_manager_id` int, -- FK -> employees.id (ID quản lý trực tiếp)
  `hr_mentor_id` int, -- FK -> employees.id (ID người hướng dẫn HR)
  `join_date` date, -- Ngày vào làm việc
  `official_start_date` date, -- Ngày ký hợp đồng chính thức
  `employment_status` varchar, -- Trạng thái (WORKING/RESIGNED/PROBATION)
  `is_deleted` tinyint, -- Trạng thái xóa mềm (LUÔN lọc is_deleted = 0)
  PRIMARY KEY (`id`)
);

CREATE TABLE `departments` (
  `id` int, -- Khóa chính, ID phòng ban
  `department_name` varchar, -- Tên phòng ban (VD: Software Development)
  `parent_department_id` int, -- FK -> departments.id (Phòng ban cha nếu lồng nhau)
  `manager_employee_id` int, -- FK -> employees.id (ID Trưởng phòng)
  `holiday_group_id` int, -- FK -> holiday_groups.id (Nhóm lịch nghỉ lễ áp dụng)
  `is_deleted` tinyint, -- Trạng thái xóa mềm (LUÔN lọc is_deleted = 0)
  PRIMARY KEY (`id`)
);

CREATE TABLE `positions` (
  `id` int, -- Khóa chính, ID chức danh/chức vụ
  `position_name` varchar, -- Tên chức danh công việc (VD: Kỹ sư Backend)
  `is_deleted` tinyint, -- Trạng thái xóa mềm (LUÔN lọc is_deleted = 0)
  PRIMARY KEY (`id`)
);

CREATE TABLE `job_grades` (
  `id` int, -- Khóa chính, ID bậc lương
  `grade_name` varchar, -- Tên bậc lương (VD: Intern, Junior, Middle, Senior, Lead)
  `min_salary` decimal, -- Lương tối thiểu của bậc này
  `max_salary` decimal, -- Lương tối đa của bậc này
  `is_deleted` tinyint, -- Trạng thái xóa mềm (LUÔN lọc is_deleted = 0)
  PRIMARY KEY (`id`)
);

CREATE TABLE `employee_salaries` (
  `id` int, -- Khóa chính
  `employee_id` int, -- FK -> employees.id (ID nhân viên)
  `job_grade_id` int, -- FK -> job_grades.id (ID bậc lương áp dụng)
  `base_salary` decimal, -- Lương cơ bản thuần (tiền đồng)
  `performance_salary` decimal, -- Lương hiệu suất / KPI
  `lunch_allowance` decimal, -- Phụ cấp ăn trưa
  `fuel_allowance` decimal, -- Phụ cấp xăng xe
  `phone_allowance` decimal, -- Phụ cấp điện thoại
  `other_allowance` decimal, -- Phụ cấp khác
  `salary_type` int, -- Loại lương (1 = Lương tháng, 2 = Lương giờ)
  `effective_from` date, -- Ngày bắt đầu áp dụng mức lương này
  `effective_to` date, -- Ngày kết thúc (NULL = hiện tại đang áp dụng)
  `salary_status` varchar, -- Trạng thái lương (ACTIVE = hiện tại, INACTIVE = lịch sử)
  `is_deleted` tinyint, -- Trạng thái xóa mềm (LUÔN lọc is_deleted = 0)
  PRIMARY KEY (`id`)
);

CREATE TABLE `payrolls` (
  `id` int, -- Khóa chính, ID bảng tính lương tổng hợp
  `payroll_month` int, -- Tháng tính lương (1-12)
  `payroll_year` int, -- Năm tính lương (VD: 2026)
  `payroll_status` varchar, -- Trạng thái (DRAFT/SUBMITTED/APPROVED)
  `payment_date` datetime, -- Ngày thực hiện thanh toán lương
  `submitted_by` int, -- FK -> employees.id (Người gửi phê duyệt)
  `approved_by` int, -- FK -> employees.id (Người duyệt bảng lương)
  `is_deleted` tinyint, -- Trạng thái xóa mềm (LUÔN lọc is_deleted = 0)
  PRIMARY KEY (`id`)
);

CREATE TABLE `payroll_details` (
  `id` int, -- Khóa chính, ID chi tiết lương từng nhân viên
  `payroll_id` int, -- FK -> payrolls.id (ID bảng lương tháng tổng hợp)
  `employee_id` int, -- FK -> employees.id (ID nhân viên nhận lương)
  `working_days` decimal, -- Số ngày công thực tế làm việc trong tháng
  `standard_days` decimal, -- Số ngày công tiêu chuẩn trong tháng (ngày công chuẩn)
  `base_salary` decimal, -- Lương cơ bản thực tế tính theo số công
  `performance_salary` decimal, -- Lương hiệu suất / KPI thực tế
  `overtime_pay` decimal, -- Tổng số tiền làm tăng ca/OT nhận được
  `bonus` decimal, -- Tiền thưởng thêm (thưởng dự án, thưởng nóng...)
  `allowance_amount` decimal, -- Tổng các khoản phụ cấp cộng vào lương
  `union_fee` decimal, -- Lệ phí công đoàn khấu trừ
  `social_insurance` decimal, -- Tiền bảo hiểm xã hội cá nhân đóng (BHXH)
  `health_insurance` decimal, -- Tiền bảo hiểm y tế cá nhân đóng (BHYT)
  `unemployment_insurance` decimal, -- Tiền bảo hiểm thất nghiệp cá nhân đóng (BHTN)
  `dependent_count` int, -- Số người phụ thuộc để giảm trừ gia cảnh thuế
  `family_deduction` decimal, -- Số tiền giảm trừ gia cảnh thuế TNCN
  `tax_deduction` decimal, -- Tiền thuế thu nhập cá nhân bị khấu trừ (Thuế TNCN)
  `insurance_deduction` decimal, -- Tổng tiền khấu trừ bảo hiểm cá nhân đóng
  `total_deduction` decimal, -- Tổng các khoản khấu trừ giảm lương (BHXH + BHYT + Thuế + Công đoàn)
  `total_gross_income` decimal, -- Tổng thu nhập gộp (Lương gộp trước thuế/bảo hiểm)
  `net_salary` decimal, -- Lương thực nhận (Thực lĩnh của nhân viên = Gross - Khấu trừ)
  `total_hr_cost` decimal, -- Tổng chi phí nhân sự thực tế công ty phải trả cho nhân viên này
  `official_days` decimal, -- Ngày công chính thức
  `probation_days` decimal, -- Ngày công thử việc
  `annual_leave_days` decimal, -- Số ngày nghỉ phép năm hưởng lương thực tế
  `unpaid_leave_days` decimal, -- Số ngày nghỉ không lương thực tế
  `total_ot_hours` decimal, -- Tổng số giờ tăng ca/OT thực tế trong tháng
  `ot_weekday` decimal, -- Giờ OT ngày thường
  `ot_weekend` decimal, -- Giờ OT cuối tuần
  `ot_holiday` decimal, -- Giờ OT ngày lễ
  `is_deleted` tinyint, -- Trạng thái xóa mềm (LUÔN lọc is_deleted = 0)
  PRIMARY KEY (`id`)
);

CREATE TABLE `contracts` (
  `id` int, -- Khóa chính, ID hợp đồng
  `employee_id` int, -- FK -> employees.id (ID nhân viên ký hợp đồng)
  `contract_number` varchar, -- Số hợp đồng lao động
  `contract_type` varchar, -- Loại hợp đồng (permanent=Vô thời hạn, temporary=Có thời hạn, probation=Thử việc)
  `start_date` date, -- Ngày bắt đầu hợp đồng hiệu lực
  `end_date` date, -- Ngày hết hạn hợp đồng (NULL = không thời hạn)
  `working_hours` decimal, -- Số giờ làm việc quy định trên tuần (VD: 40.00)
  `contract_status` varchar, -- Trạng thái (ACTIVE=Đang hiệu lực, EXPIRED=Hết hạn)
  `signed_date` date, -- Ngày ký kết hợp đồng
  `is_deleted` tinyint, -- Trạng thái xóa mềm (LUÔN lọc is_deleted = 0)
  PRIMARY KEY (`id`)
);

CREATE TABLE `attendance_records` (
  `id` int, -- Khóa chính, ID bản ghi chấm công thô gốc từ máy/GPS
  `employee_id` int, -- FK -> employees.id (ID nhân viên chấm công)

  `late_minutes` int, -- Số phút đi muộn (nếu có)
  `early_leave_minutes` int, -- Số phút về sớm (nếu có)
  `overtime_minutes` int, -- Số phút tăng ca trong ngày chấm công
  `total_work_minutes` int, -- Tổng số phút làm việc thực tế trong ngày
  `working_shift_id` int, -- FK -> working_shifts.id (Mã ca làm việc được phân)
  `assignment_id` int, -- FK -> shift_assignments.id (Mã phân ca tương ứng)

  PRIMARY KEY (`id`)
);

  PRIMARY KEY (`id`)

CREATE TABLE `penalties` (

CREATE TABLE `overtime_rules` (

CREATE TABLE `requests` (
  `id` int, -- Khóa chính, ID đơn từ hành chính
  `request_code` varchar, -- Mã đơn từ duy nhất (REQ-2026-xxx)
  `request_group_id` int, -- FK -> request_groups.id (Mã nhóm đơn: 1=LEAVE nghỉ phép, 2=OVERTIME tăng ca, 3=ATTENDANCE giải trình chấm công)
  `request_type_id` int, -- FK -> request_types.id (Mã loại đơn)
  `employee_id` int, -- FK -> employees.id (ID nhân viên làm đơn)
  `start_date` date, -- Ngày bắt đầu đơn hiệu lực (ngày nghỉ phép từ, ngày OT...)
  `end_date` date, -- Ngày kết thúc đơn
  `quantity` decimal, -- Số lượng đơn vị xin (Ví dụ: số ngày xin nghỉ 1.00; số giờ xin OT 3.00)
  `unit` varchar, -- Đơn vị lượng xin (day = ngày, hour = giờ)
  `status` varchar, -- Trạng thái phê duyệt đơn (DRAFT/SUBMITTED/APPROVED/REJECTED/CANCELLED)
  `is_deleted` tinyint, -- Trạng thái xóa mềm (LUÔN lọc is_deleted = 0)
  PRIMARY KEY (`id`)
);

CREATE TABLE `request_types` (

  PRIMARY KEY (`id`)

CREATE TABLE `request_groups` (
