-- =============================================================================
-- SMARTHR DATABASE DESCRIPTION
-- Mô tả chi tiết các bảng trong hệ thống Quản lý Nhân sự SkyBreath SmartHR.
-- Dùng để hỗ trợ AI sinh câu lệnh SQL chính xác khi nhận câu hỏi từ người dùng.
-- LƯU Ý CHUNG:
--   - Hầu hết các bảng đều có: is_deleted=0 (chưa xóa), is_deleted=1 (đã xóa mềm).
--   - LUÔN thêm điều kiện is_deleted = 0 khi truy vấn để tránh lấy dữ liệu đã xóa.
--   - Cột is_active (tinyint 0/1): 0=tắt, 1=bật/đang hoạt động.
-- =============================================================================


-- =============================================================================
-- BẢNG: employees
-- MÔ TẢ: Bảng trung tâm - lưu thông tin toàn bộ nhân viên.
-- DÙNG CHO CÂU HỎI: thông tin nhân viên, tên, phòng ban, chức vụ, ngày vào làm.
-- CÁC PHÉP JOIN PHỔ BIẾN:
--   - JOIN departments d ON e.department_id = d.id → lấy tên phòng ban
--   - JOIN positions p ON e.position_id = p.id → lấy chức danh
--   - JOIN job_grades jg ON e.job_grade_id = jg.id → lấy bậc lương
--   - JOIN employees mgr ON e.direct_manager_id = mgr.id → lấy tên quản lý trực tiếp
-- VÍ DỤ HỎI: "Nhân viên A đang làm ở phòng ban nào?"
--   → SELECT e.full_name, d.department_name FROM employees e JOIN departments d ON e.department_id = d.id WHERE e.full_name LIKE '%A%' AND e.is_deleted = 0
-- =============================================================================
CREATE TABLE `employees` (
  `id` int NOT NULL AUTO_INCREMENT,                          -- Khóa chính, ID nhân viên
  `employee_code` varchar(20) DEFAULT NULL,                  -- Mã nhân viên (VD: NV001)
  `user_id` int DEFAULT NULL,                                -- FK → users.id (tài khoản đăng nhập)
  `full_name` varchar(255) NOT NULL,                         -- Họ tên đầy đủ (dùng để tìm kiếm theo tên)
  `date_of_birth` date DEFAULT NULL,                         -- Ngày sinh
  `gender` varchar(255) DEFAULT NULL,                        -- Giới tính (MALE/FEMALE/OTHER)
  `marital_status` varchar(255) DEFAULT NULL,                -- Tình trạng hôn nhân
  `national_id` varchar(255) DEFAULT NULL,                   -- Số CCCD/CMND
  `tax_code` varchar(255) DEFAULT NULL,                      -- Mã số thuế cá nhân
  `personal_email` varchar(255) DEFAULT NULL,                -- Email cá nhân
  `company_email` varchar(255) DEFAULT NULL,                 -- Email công ty
  `phone_number` varchar(255) DEFAULT NULL,                  -- Số điện thoại
  `department_id` int DEFAULT NULL,                          -- FK → departments.id (phòng ban hiện tại)
  `position_id` int DEFAULT NULL,                            -- FK → positions.id (chức danh hiện tại)
  `job_grade_id` int DEFAULT NULL,                           -- FK → job_grades.id (bậc lương)
  `direct_manager_id` int DEFAULT NULL,                      -- FK → employees.id (quản lý trực tiếp)
  `hr_mentor_id` int DEFAULT NULL,                           -- FK → employees.id (HR mentor kèm cặp)
  `join_date` date DEFAULT NULL,                             -- Ngày vào làm việc thực tế
  `official_start_date` date DEFAULT NULL,                   -- Ngày chính thức (sau thử việc)
  `employment_status` varchar(255) DEFAULT NULL,             -- Trạng thái (WORKING, RESIGNED, PROBATION…)
  `plan_id` int DEFAULT NULL,                                -- FK → onboarding_plans.id (kế hoạch onboarding)
  `is_deleted` tinyint NOT NULL DEFAULT '0',                 -- Xóa mềm (0=tồn tại, 1=đã xóa)
  `created_at` datetime(6) NOT NULL DEFAULT CURRENT_TIMESTAMP(6),
  `updated_at` datetime(6) NOT NULL DEFAULT CURRENT_TIMESTAMP(6) ON UPDATE CURRENT_TIMESTAMP(6),
  PRIMARY KEY (`id`),
  CONSTRAINT `FK_678a3540f843823784b0fe4a4f2` FOREIGN KEY (`department_id`) REFERENCES `departments` (`id`),
  CONSTRAINT `FK_8b14204e8af5e371e36b8c11e1b` FOREIGN KEY (`position_id`) REFERENCES `positions` (`id`),
  CONSTRAINT `FK_b4f73f38d40465a2dc06466b9cd` FOREIGN KEY (`job_grade_id`) REFERENCES `job_grades` (`id`),
  CONSTRAINT `FK_ef99e9967917b947ce02019fbc5` FOREIGN KEY (`direct_manager_id`) REFERENCES `employees` (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;


-- =============================================================================
-- BẢNG: departments
-- MÔ TẢ: Cơ cấu phòng ban của công ty (có thể lồng cấp cha-con).
-- DÙNG CHO CÂU HỎI: nhân viên thuộc phòng nào, phòng ban nào có bao nhiêu người.
-- CÁC PHÉP JOIN PHỔ BIẾN:
--   - JOIN employees e ON e.department_id = d.id → danh sách NV thuộc phòng
--   - JOIN employees mgr ON d.manager_employee_id = mgr.id → lấy tên trưởng phòng
--   - LEFT JOIN departments parent ON d.parent_department_id = parent.id → phòng ban cấp cha
-- =============================================================================
CREATE TABLE `departments` (
  `id` int NOT NULL AUTO_INCREMENT,
  `department_name` varchar(255) NOT NULL,                   -- Tên phòng ban
  `parent_department_id` int DEFAULT NULL,                   -- FK → departments.id (phòng ban cha, nếu có)
  `manager_employee_id` int DEFAULT NULL,                    -- FK → employees.id (trưởng phòng)
  `holiday_group_id` int DEFAULT NULL,                       -- FK → holiday_groups.id (nhóm lịch nghỉ lễ)
  `is_deleted` tinyint NOT NULL DEFAULT '0',
  `created_at` datetime(6) NOT NULL DEFAULT CURRENT_TIMESTAMP(6),
  `updated_at` datetime(6) NOT NULL DEFAULT CURRENT_TIMESTAMP(6) ON UPDATE CURRENT_TIMESTAMP(6),
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;


-- =============================================================================
-- BẢNG: positions
-- MÔ TẢ: Danh mục các chức danh công việc (Kỹ sư phần mềm, HR, Kế toán…).
-- DÙNG CHO CÂU HỎI: nhân viên giữ chức danh gì, danh sách chức danh.
-- JOIN: employees e JOIN positions p ON e.position_id = p.id
-- =============================================================================
CREATE TABLE `positions` (
  `id` int NOT NULL AUTO_INCREMENT,
  `position_name` varchar(255) NOT NULL,                     -- Tên chức danh (VD: "Kỹ sư Backend")
  `is_deleted` tinyint NOT NULL DEFAULT '0',
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;


-- =============================================================================
-- BẢNG: job_grades
-- MÔ TẢ: Bậc lương / cấp bậc (G1-G5 tương ứng khoảng lương min-max).
-- DÙNG CHO CÂU HỎI: nhân viên thuộc cấp bậc lương nào, khoảng lương của bậc đó.
-- JOIN: employees e JOIN job_grades jg ON e.job_grade_id = jg.id
-- =============================================================================
CREATE TABLE `job_grades` (
  `id` int NOT NULL AUTO_INCREMENT,
  `grade_name` varchar(255) NOT NULL,                        -- Tên bậc dùng chung cho tất cả phòng ban (VD: G1, G2, Senior)
  `min_salary` decimal(15,2) NOT NULL,                       -- Lương tối thiểu của bậc
  `max_salary` decimal(15,2) NOT NULL,                       -- Lương tối đa của bậc
  `is_deleted` tinyint NOT NULL DEFAULT '0',
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;


-- =============================================================================
-- BẢNG: employee_salaries
-- MÔ TẢ: Bảng lương cơ bản và phụ cấp của từng nhân viên theo từng giai đoạn.
-- ĐẶC BIỆT QUAN TRỌNG: Một nhân viên có thể có nhiều dòng lương (lịch sử tăng lương).
--   Để lấy mức lương HIỆN TẠI: lọc salary_status = 'ACTIVE' HOẶC theo effective_from/effective_to.
-- DÙNG CHO CÂU HỎI: lương cơ bản của A là bao nhiêu, tổng thu nhập, phụ cấp.
-- PHÉP JOIN ĐỂ LẤY LƯƠNG HIỆN TẠI:
--   SELECT e.full_name, es.base_salary, es.performance_salary, ...
--   FROM employees e JOIN employee_salaries es ON es.employee_id = e.id AND es.salary_status = 'ACTIVE' AND es.is_deleted = 0
--   WHERE e.full_name LIKE '%Tên%' AND e.is_deleted = 0
-- =============================================================================
CREATE TABLE `employee_salaries` (
  `id` int NOT NULL AUTO_INCREMENT,
  `employee_id` int NOT NULL,                                -- FK → employees.id
  `job_grade_id` int DEFAULT NULL,                           -- FK → job_grades.id
  `base_salary` decimal(15,2) NOT NULL,                      -- Lương cơ bản (tiền thuần)
  `performance_salary` decimal(15,2) NOT NULL DEFAULT '0',   -- Lương hiệu suất / KPI
  `lunch_allowance` decimal(15,2) NOT NULL DEFAULT '0',      -- Phụ cấp ăn trưa
  `fuel_allowance` decimal(15,2) NOT NULL DEFAULT '0',       -- Phụ cấp xăng xe
  `phone_allowance` decimal(15,2) NOT NULL DEFAULT '0',      -- Phụ cấp điện thoại
  `other_allowance` decimal(15,2) NOT NULL DEFAULT '0',      -- Phụ cấp khác
  `salary_type` int NOT NULL DEFAULT '1',                    -- Loại lương (1=tháng, 2=giờ)
  `effective_from` date DEFAULT NULL,                        -- Ngày bắt đầu áp dụng mức lương này
  `effective_to` date DEFAULT NULL,                          -- Ngày kết thúc (NULL = hiện tại)
  `salary_status` varchar(255) NOT NULL DEFAULT 'ACTIVE',    -- Trạng thái: ACTIVE=đang áp dụng, INACTIVE=đã thay thế
  `is_deleted` tinyint NOT NULL DEFAULT '0',
  PRIMARY KEY (`id`),
  CONSTRAINT `FK_5cee264c9d34d5836769435cb0c` FOREIGN KEY (`employee_id`) REFERENCES `employees` (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;
-- TỔNG LƯƠNG GỘP = base_salary + performance_salary + lunch_allowance + fuel_allowance + phone_allowance + other_allowance


-- =============================================================================
-- BẢNG: contracts
-- MÔ TẢ: Hợp đồng lao động của nhân viên.
-- DÙNG CHO CÂU HỎI: loại hợp đồng, ngày ký, trạng thái hợp đồng.
-- JOIN: employees e JOIN contracts c ON c.employee_id = e.id AND c.is_deleted = 0
-- =============================================================================
CREATE TABLE `contracts` (
  `id` int NOT NULL AUTO_INCREMENT,
  `employee_id` int NOT NULL,                                -- FK → employees.id
  `contract_number` varchar(255) NOT NULL,                   -- Số hợp đồng
  `contract_type` varchar(255) DEFAULT NULL,                 -- Loại hợp đồng (FULL_TIME, PART_TIME, PROBATION…)
  `start_date` date DEFAULT NULL,                            -- Ngày bắt đầu hợp đồng
  `end_date` date DEFAULT NULL,                              -- Ngày kết thúc (NULL = vô thời hạn)
  `working_hours` decimal(5,2) DEFAULT NULL,                 -- Số giờ làm việc/tuần
  `contract_status` varchar(255) DEFAULT NULL,               -- Trạng thái (ACTIVE, EXPIRED, TERMINATED)
  `signed_date` date DEFAULT NULL,                           -- Ngày ký hợp đồng
  `termination_date` date DEFAULT NULL,                      -- Ngày chấm dứt (nếu bị chấm dứt)
  `termination_reason` varchar(255) DEFAULT NULL,            -- Lý do chấm dứt
  `is_deleted` tinyint NOT NULL DEFAULT '0',
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;


-- =============================================================================
-- BẢNG: attendance_records
-- MÔ TẢ: Dữ liệu chấm công thô từ thiết bị (face/GPS/QR/thủ công).
-- DÙNG CHO CÂU HỎI: giờ check-in/out, đi muộn bao nhiêu phút, trạng thái chấm công.
-- LƯU Ý: Đây là dữ liệu gốc chưa xử lý. Để tính tổng công tháng, ưu tiên dùng processed_attendance_records.
-- JOIN: employees e JOIN attendance_records ar ON ar.employee_id = e.id AND ar.is_deleted = 0
--       JOIN shift_schedules ss ON ar.shift_schedule_id = ss.id
-- =============================================================================
CREATE TABLE `attendance_records` (
  `id` int NOT NULL AUTO_INCREMENT,
  `employee_id` int NOT NULL,                                -- FK → employees.id
  `work_date` date DEFAULT NULL,                             -- Ngày làm việc
  `shift_schedule_id` int DEFAULT NULL,                      -- FK → shift_schedules.id (ca cụ thể trong ngày)
  `check_in_time` datetime DEFAULT NULL,                     -- Thời điểm check-in
  `check_out_time` datetime DEFAULT NULL,                    -- Thời điểm check-out
  `attendance_status` enum('present','late','early_leave','absent','half_day') DEFAULT NULL,  -- Trạng thái: present=có mặt, late=muộn, early_leave=về sớm, absent=vắng, half_day=nửa ngày
  `attendance_type` enum('face','gps','qr','manual') NOT NULL DEFAULT 'face',  -- Hình thức chấm (khuôn mặt/GPS/QR/thủ công)
  `late_minutes` int DEFAULT NULL,                           -- Số phút đi muộn
  `early_leave_minutes` int DEFAULT NULL,                    -- Số phút về sớm
  `overtime_minutes` int DEFAULT NULL,                       -- Số phút tăng ca
  `total_work_minutes` int DEFAULT NULL,                     -- Tổng phút làm việc thực tế
  `is_deleted` tinyint NOT NULL DEFAULT '0',
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;


-- =============================================================================
-- BẢNG: processed_attendance_records
-- MÔ TẢ: Dữ liệu chấm công đã xử lý/tính toán (sau khi HR đồng bộ/chốt công).
-- ĐÂY LÀ BẢNG ƯU TIÊN khi hỏi về số công tháng, tổng công.
-- DÙNG CHO CÂU HỎI: tháng X chấm được bao nhiêu buổi/ngày công, tổng công tháng.
-- PHÉP JOIN:
--   SELECT e.full_name, par.work_value, par.work_date
--   FROM employees e JOIN processed_attendance_records par ON par.employee_id = e.id
--   WHERE e.full_name LIKE '%Tên%' AND MONTH(par.work_date) = X AND YEAR(par.work_date) = Y AND par.is_deleted = 0
-- =============================================================================
CREATE TABLE `processed_attendance_records` (
  `id` int NOT NULL AUTO_INCREMENT,
  `employee_id` int NOT NULL,                                -- FK → employees.id
  `work_date` date NOT NULL,                                 -- Ngày làm việc cụ thể
  `work_value` decimal(5,2) DEFAULT NULL,                    -- Giá trị công (1.0=cả ngày, 0.5=nửa ngày, 0=vắng)
  `attendance_status` varchar(50) DEFAULT NULL,              -- Trạng thái đã xử lý
  `shift_schedule_id` int DEFAULT NULL,                      -- FK → shift_schedules.id
  `is_deleted` tinyint NOT NULL DEFAULT '0',
  `created_at` datetime(6) NOT NULL DEFAULT CURRENT_TIMESTAMP(6),
  `updated_at` datetime(6) NOT NULL DEFAULT CURRENT_TIMESTAMP(6) ON UPDATE CURRENT_TIMESTAMP(6),
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;
-- ĐẾM SỐ BUỔI CHẤM CÔNG: COUNT(*) WHERE work_value > 0 AND MONTH(work_date)=X
-- TỔNG CÔNG: SUM(work_value) WHERE MONTH(work_date)=X


-- =============================================================================
-- BẢNG: time_sheets
-- MÔ TẢ: Bảng tổng hợp công theo tháng của mỗi nhân viên (đã được HR tạo/chốt).
-- DÙNG CHO CÂU HỎI: tổng ngày công tháng X của nhân viên A, tổng giờ OT.
-- LƯU Ý: is_locked=1 nghĩa là đã chốt công, không thể sửa.
-- JOIN: employees e JOIN time_sheets ts ON ts.employee_id = e.id AND ts.is_deleted = 0
-- =============================================================================
CREATE TABLE `time_sheets` (
  `id` int NOT NULL AUTO_INCREMENT,
  `employee_id` int NOT NULL,                                -- FK → employees.id
  `month` int NOT NULL,                                      -- Tháng (1-12)
  `year` int NOT NULL,                                       -- Năm (VD: 2026)
  `total_working_days` decimal(5,2) DEFAULT NULL,            -- Tổng ngày công trong tháng
  `total_working_hours` decimal(10,2) DEFAULT NULL,          -- Tổng giờ làm việc
  `overtime_hours` decimal(10,2) DEFAULT NULL,               -- Tổng giờ tăng ca
  `is_locked` tinyint NOT NULL DEFAULT '0',                  -- 0=chưa chốt, 1=đã chốt/khóa
  `is_deleted` tinyint NOT NULL DEFAULT '0',
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;


-- =============================================================================
-- BẢNG: leave_types
-- MÔ TẢ: Danh mục các loại nghỉ phép (Phép năm, Phép bệnh, Phép không lương…).
-- DÙNG CHO CÂU HỎI: các loại phép, phép có lương hay không.
-- =============================================================================
CREATE TABLE `leave_types` (
  `id` int NOT NULL AUTO_INCREMENT,
  `leave_type_name` varchar(255) NOT NULL,                   -- Tên loại phép (VD: "Phép năm có lương")
  `is_paid` tinyint NOT NULL DEFAULT '0',                    -- 0=không lương, 1=có lương
  `is_deleted` tinyint NOT NULL DEFAULT '0',
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;


-- =============================================================================
-- BẢNG: leave_policies
-- MÔ TẢ: Chính sách phép năm (số ngày phép tối đa/năm) theo từng loại phép.
-- DÙNG CHO CÂU HỎI: tổng số ngày phép được cấp là bao nhiêu.
-- JOIN: leave_types lt JOIN leave_policies lp ON lp.leave_type_id = lt.id AND lp.is_deleted = 0
-- =============================================================================
CREATE TABLE `leave_policies` (
  `id` int NOT NULL AUTO_INCREMENT,
  `leave_type_id` int NOT NULL,                              -- FK → leave_types.id
  `policy_name` varchar(255) NOT NULL,                       -- Tên chính sách
  `days_per_year` int NOT NULL,                              -- Số ngày phép được cấp mỗi năm
  `is_deleted` tinyint NOT NULL DEFAULT '0',
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;


-- =============================================================================
-- BẢNG: leave_balances
-- MÔ TẢ: Theo dõi số ngày phép đã sử dụng của từng nhân viên theo năm.
-- CÔNG THỨC NGÀY PHÉP CÒN LẠI:
--   Ngày phép còn lại = lp.days_per_year - lb.used_days
-- DÙNG CHO CÂU HỎI: nhân viên A còn bao nhiêu ngày phép có lương?
-- PHÉP JOIN ĐẦY ĐỦ:
--   SELECT e.full_name, lt.leave_type_name, lp.days_per_year, lb.used_days,
--          (lp.days_per_year - lb.used_days) AS remaining_days
--   FROM employees e
--   JOIN leave_balances lb ON lb.employee_id = e.id AND lb.is_deleted = 0
--   JOIN leave_types lt ON lt.id = lb.leave_type_id AND lt.is_deleted = 0
--   JOIN leave_policies lp ON lp.leave_type_id = lt.id AND lp.is_deleted = 0
--   WHERE e.full_name LIKE '%Tên%' AND e.is_deleted = 0 AND lb.year = YEAR(CURDATE())
-- LƯU Ý: Không có cột total_days trong bảng này. Tổng phép từ leave_policies.days_per_year.
-- =============================================================================
CREATE TABLE `leave_balances` (
  `id` int NOT NULL AUTO_INCREMENT,
  `employee_id` int NOT NULL,                                -- FK → employees.id
  `leave_type_id` int NOT NULL,                              -- FK → leave_types.id
  `year` int NOT NULL,                                       -- Năm tính phép (VD: 2026)
  `used_days` decimal(5,2) NOT NULL DEFAULT '0',             -- Số ngày phép đã dùng
  `is_deleted` tinyint NOT NULL DEFAULT '0',
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;


-- =============================================================================
-- BẢNG: working_shifts
-- MÔ TẢ: Danh mục các ca làm việc (Ca sáng 8h-17h, Ca tối 18h-22h…).
-- DÙNG CHO CÂU HỎI: các ca làm việc hiện có, giờ vào-ra của từng ca.
-- =============================================================================
CREATE TABLE `working_shifts` (
  `id` int NOT NULL AUTO_INCREMENT,
  `shift_name` varchar(255) NOT NULL,                        -- Tên ca (VD: "Ca Sáng", "Ca Hành Chính")
  `start_time` time DEFAULT NULL,                            -- Giờ bắt đầu ca
  `end_time` time DEFAULT NULL,                              -- Giờ kết thúc ca
  `break_start_time` time DEFAULT NULL,                      -- Giờ bắt đầu nghỉ giữa ca
  `break_end_time` time DEFAULT NULL,                        -- Giờ kết thúc nghỉ giữa ca
  `group_id` int DEFAULT NULL,                               -- FK → shift_groups.id (nhóm ca)
  `is_deleted` tinyint NOT NULL DEFAULT '0',
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;


-- =============================================================================
-- BẢNG: shift_groups
-- MÔ TẢ: Nhóm ca (VD: Nhóm Ca Hành Chính, Nhóm Ca Luân Phiên).
-- =============================================================================
CREATE TABLE `shift_groups` (
  `id` int NOT NULL AUTO_INCREMENT,
  `group_name` varchar(255) NOT NULL,                        -- Tên nhóm ca
  `description` varchar(255) DEFAULT NULL,
  `status` varchar(20) NOT NULL DEFAULT 'active',            -- active/inactive
  `is_deleted` tinyint NOT NULL DEFAULT '0',
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;


-- =============================================================================
-- BẢNG: shift_assignments
-- MÔ TẢ: Phân ca - giao ca làm việc cho nhân viên hoặc phòng ban.
-- ĐẶC BIỆT: Có thể phân theo nhân viên (employee_id) HOẶC phòng ban (department_id).
--   Cột employee_ids và department_ids (text) dùng để lưu nhiều ID dạng JSON string.
-- DÙNG CHO CÂU HỎI: nhân viên A đang làm ca gì, được phân ca nào.
-- PHÉP JOIN ĐỂ LẤY CA CỦA NHÂN VIÊN:
--   SELECT e.full_name, ws.shift_name, ws.start_time, ws.end_time, sa.effective_from, sa.effective_to
--   FROM employees e
--   JOIN shift_assignments sa ON sa.employee_id = e.id AND sa.is_deleted = 0
--   JOIN working_shifts ws ON ws.id = sa.shift_id AND ws.is_deleted = 0
--   WHERE e.full_name LIKE '%Tên%' AND e.is_deleted = 0
--     AND (sa.effective_to IS NULL OR sa.effective_to >= CURDATE())
-- LƯU Ý: Nếu employee_id là NULL, check thêm department_id hoặc employee_ids.
-- =============================================================================
CREATE TABLE `shift_assignments` (
  `id` int NOT NULL AUTO_INCREMENT,
  `assignment_name` varchar(255) DEFAULT NULL,               -- Tên bản phân ca
  `employee_id` int DEFAULT NULL,                            -- FK → employees.id (phân cho 1 nhân viên cụ thể)

-- =============================================================================
-- BẢNG: requests
-- MÔ TẢ: Bảng tổng hợp tất cả các đơn từ của nhân viên (nghỉ phép, OT, công tác…).
-- TRẠNG THÁI: DRAFT→SUBMITTED→APPROVED/REJECTED/CANCELLED
-- DÙNG CHO CÂU HỎI: đơn nghỉ phép đang chờ duyệt, đơn OT tháng này.
-- QUAN TRỌNG: Bảng requests dùng cột `status` (DRAFT/SUBMITTED/APPROVED/REJECTED/CANCELLED).
--             Dùng cột `quantity` để biết số ngày/giờ xin.
-- JOIN với leave_types thông qua request_types để biết loại đơn.
-- =============================================================================
CREATE TABLE `requests` (
  `id` int NOT NULL AUTO_INCREMENT,
  `request_code` varchar(50) DEFAULT NULL,                   -- Mã đơn duy nhất (VD: REQ-2026-001)
  `request_group_id` int NOT NULL,                           -- FK → request_groups.id
  `request_type_id` int NOT NULL,                            -- FK → request_types.id (loại đơn)
  `employee_id` int NOT NULL,                                -- FK → employees.id (người nộp đơn)
  `start_date` date DEFAULT NULL,                            -- Ngày bắt đầu (VD: nghỉ từ ngày)
  `end_date` date DEFAULT NULL,                              -- Ngày kết thúc
  `quantity` decimal(10,2) DEFAULT NULL,                     -- Số lượng (ngày/giờ xin)
  `unit` varchar(50) DEFAULT NULL,                           -- Đơn vị (day/hour)
  `description` text,                                        -- Lý do/mô tả
  `status` varchar(50) NOT NULL DEFAULT 'DRAFT',             -- Trạng thái: DRAFT/SUBMITTED/APPROVED/REJECTED/CANCELLED
  `submitted_at` datetime DEFAULT NULL,
  `approved_at` datetime DEFAULT NULL,
  `rejected_at` datetime DEFAULT NULL,
  `is_worked_time` tinyint NOT NULL DEFAULT '0',             -- 1=là giờ làm bù/OT, 0=không
  `is_deleted` tinyint NOT NULL DEFAULT '0',
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;


-- =============================================================================
